  try {

}

function startQuestionTimer() {
  currentQStartTime = Date.now();
  clearInterval(qTimerInterval);
  updateCurrentQTimeDisplay();
  qTimerInterval = setInterval(updateCurrentQTimeDisplay, 1000);
}

function commitCurrentQTime() {
  if (currentQStartTime > 0) {
    const elapsedSecs = Math.max(0, Math.round((Date.now() - currentQStartTime) / 1000));
    userTimes[currentQIndex] = (userTimes[currentQIndex] || 0) + elapsedSecs;
    currentQStartTime = Date.now();
  }
}

function updateCurrentQTimeDisplay() {
  const currentElapsed = Math.round((Date.now() - currentQStartTime) / 1000);
  const totalOnThisQ = (userTimes[currentQIndex] || 0) + currentElapsed;
  const m = Math.floor(totalOnThisQ / 60);
  const s = String(totalOnThisQ % 60).padStart(2, "0");
  const el = $("testQTimeCounter");
  if (el) el.textContent = `${m}:${s}`;
}

function testTimerTick() {
  const remaining = Math.max(0, Math.ceil((testEndTime - Date.now()) / 1000));
  updateTestTimerDisplay(remaining);
  if (remaining === 0) {
    commitCurrentQTime();
    clearInterval(testTimerInterval);
    clearInterval(qTimerInterval);
    alert("Time is up! Your 5 minutes have concluded. Prepwise will now compile your results.");
    submitTest(true);
  }
}

function updateTestTimerDisplay(leftSecs = null) {
  if (leftSecs === null) {
    leftSecs = Math.max(0, Math.ceil((testEndTime - Date.now()) / 1000));
  }
  const m = String(Math.floor(leftSecs / 60)).padStart(2, "0");
  const s = String(leftSecs % 60).padStart(2, "0");

  const timerPill = $("testTimer");
  if (timerDigits) timerDigits.textContent = `${m}:${s}`;
  if (timerPill) {
    timerPill.classList.toggle("low", leftSecs <= 60);
  }
}

function submitTest(isAuto = false) {
  clearInterval(testTimerInterval);
  clearInterval(qTimerInterval);
  testTimerInterval = null;
  qTimerInterval = null;

  let correctCount = 0;
  let wrongCount = 0;
  let skippedCount = 0;

  // Track per-test topic performance to provide immediate diagnostic insights
  const runTopicPerformance = {};

  activeQuestions.forEach((q, i) => {
    const ans = userAnswers[i];
    const isCorrect = ans === q.a;
    const isSkipped = ans === null;

    if (isSkipped) skippedCount++;
    else if (isCorrect) correctCount++;
    else wrongCount++;

    if (!runTopicPerformance[q.topic]) {
      runTopicPerformance[q.topic] = { total: 0, correct: 0, wrong: 0, skipped: 0 };
    }
    runTopicPerformance[q.topic].total++;
    if (isCorrect) runTopicPerformance[q.topic].correct++;
    else if (isSkipped) runTopicPerformance[q.topic].skipped++;
    else runTopicPerformance[q.topic].wrong++;
  });

  const totalTimeSecs = Math.min(TEST_DURATION_SECONDS, Math.round((Date.now() - testStartTime) / 1000));
  const avgTimePerQ = Math.round(totalTimeSecs / activeQuestions.length);
  const accuracyPct = Math.round((correctCount / activeQuestions.length) * 100);

  // Compile detailed result object
  lastSubmittedResult = {
    id: "attempt_" + Date.now(),
    date: new Date().toLocaleString(),
    timestamp: Date.now(),
    categoryKey: currentCategoryKey,
    categoryName: QUESTION_BANK[currentCategoryKey].name,
    totalQuestions: activeQuestions.length,
    correctCount,
    wrongCount,
    skippedCount,
    totalTimeSecs,
    avgTimePerQ,
    accuracyPct,
    questions: activeQuestions,
    userAnswers: [...userAnswers],
    userTimes: [...userTimes],
    topicBreakdown: runTopicPerformance
  };

  // Persist to Lifetime Topic Mastery Matrix and Test History
  recordAttemptInStorage(lastSubmittedResult);

  // Render Result Screen
  renderResultScreen(lastSubmittedResult, isAuto);
  showView("result");
}

function recordAttemptInStorage(res) {
  // 1. Update Test History
  const history = getStoredJson(STORAGE_KEYS.HISTORY, []);
  history.unshift({
    id: res.id,
    date: res.date,
    categoryName: res.categoryName,
    categoryKey: res.categoryKey,
    scoreText: `${res.correctCount}/${res.totalQuestions}`,
    accuracyPct: res.accuracyPct,
    timeSecs: res.totalTimeSecs,
    weakTopics: Object.entries(res.topicBreakdown)
      .filter(([_, s]) => s.correct < s.total)
      .map(([t]) => t)
  });
  setStoredJson(STORAGE_KEYS.HISTORY, history.slice(0, 30)); // Keep last 30 tests

  // 2. Update Topic Mastery Matrix
  const topicStats = getStoredJson(STORAGE_KEYS.TOPIC_STATS, {});
  res.questions.forEach((q, i) => {
    const ans = res.userAnswers[i];
    const isCorrect = ans === q.a;
    const isSkipped = ans === null;

    if (!topicStats[q.topic]) {
      topicStats[q.topic] = {
        categoryKey: res.categoryKey,
        attempted: 0,
        correct: 0,
        wrong: 0,
        skipped: 0,
        totalTimeSec: 0
      };
    }
    topicStats[q.topic].attempted++;
    topicStats[q.topic].totalTimeSec += (res.userTimes[i] || 0);
    if (isCorrect) topicStats[q.topic].correct++;
    else if (isSkipped) topicStats[q.topic].skipped++;
    else topicStats[q.topic].wrong++;
  });
  setStoredJson(STORAGE_KEYS.TOPIC_STATS, topicStats);

  // Update Nav Badge
  updateWeakNavBadge();
}

function computeWeakTopics() {
  const topicStats = getStoredJson(STORAGE_KEYS.TOPIC_STATS, {});
  const weakList = [];

  Object.entries(topicStats).forEach(([topic, stat]) => {
    if (stat.attempted >= 1) {
      const acc = Math.round((stat.correct / stat.attempted) * 100);
      if (acc < 60) {
        weakList.push({
          topic,
          accuracy: acc,
          attempted: stat.attempted,
          correct: stat.correct,
          wrong: stat.wrong,
          skipped: stat.skipped,
          advice: getTopicRemedialAdvice(topic)
        });
      }
    }
  });

  // Sort by lowest accuracy first
  return weakList.sort((a, b) => a.accuracy - b.accuracy);
}

function computeStrongTopics() {
  const topicStats = getStoredJson(STORAGE_KEYS.TOPIC_STATS, {});
  const strongList = [];

  Object.entries(topicStats).forEach(([topic, stat]) => {
    if (stat.attempted >= 1) {
      const acc = Math.round((stat.correct / stat.attempted) * 100);
      if (acc >= 75) {
        strongList.push({
          topic,
          accuracy: acc,
          attempted: stat.attempted,
          correct: stat.correct
        });
      }
    }
  });

  return strongList.sort((a, b) => b.accuracy - a.accuracy);
}

  }

function toggleBookmark(questionId) {
  let bookmarks = getStoredJson(STORAGE_KEYS.BOOKMARKS, []);
  if (bookmarks.includes(questionId)) {
    bookmarks = bookmarks.filter(id => id !== questionId);
  } else {
    bookmarks.push(questionId);
  }
  setStoredJson(STORAGE_KEYS.BOOKMARKS, bookmarks);

window.addEventListener("keydown", (e) => {
  // Only trigger keyboard shortcuts when test is actively running
  if ($("view-test").classList.contains("hidden")) return;
  if (e.target.tagName === "INPUT" || e.target.tagName === "TEXTAREA" || e.target.tagName === "SELECT") return;

  const key = e.key.toUpperCase();
  if (["1", "2", "3", "4"].includes(key)) {
    selectOption(parseInt(key) - 1);
  } else if (["A", "B", "C", "D"].includes(key)) {
    selectOption("ABCD".indexOf(key));
  } else if (key === "ARROWLEFT" || key === "P") {
    navigateQuestion(-1);
  } else if (key === "ARROWRIGHT" || key === "N") {
    navigateQuestion(1);
  } else if (key === "M") {
    toggleMarkCurrentQuestion();
  }
});

// Utility to prevent XSS in dynamic text
function escapeHtml(str) {
  if (typeof str !== "string") return str;
  return str.replace(/[&<>'"]/g, tag => ({
    "&": "&amp;",
    "<": "&lt;",
    ">": "&gt;",
    "'": "&#39;",
    '"': "&quot;"
  }[tag] || tag));
}
